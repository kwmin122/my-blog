<objective>
Implement the accessibility DOM mirror that exposes all world text content to screen readers. This is a plain React DOM component (no R3F) that sits as a sibling of the canvas, renders static sr-only divs for every post's title/excerpt/visuals, and maintains an aria-live region announcing the currently active waypoint. Independently implementable from Plan 01 — the component accepts props so it can work with a hardcoded post list initially and be updated to real data when Plan 01 posts land.

Requirements fulfilled: A11Y-02
Phase goal contribution: Closes the screen reader accessibility gap for the 3D world. World text rendered inside R3F's Html layer is invisible to AT; this mirror makes it fully accessible without touching the 3D layer.
Complexity: M
</objective>

  <action>
    Create `components/world/WorldSRMirror.tsx` as a `'use client'` component. This is a plain DOM component — no R3F, no canvas. It is placed as a sibling of `<canvas>` (D-03a: NOT inside `<Html>` 3D layer).

    **Interface shape:**

    ```typescript
    export interface PostSRData {
      slug: string
      title: string
      excerpt: string
      visuals: { id: string; alt: string }[]
    }
    ```

    **Component logic:**
    1. Subscribes to `worldStore.activeWaypoint` to get the currently active slug.
    2. Renders a wrapping div with `className="sr-only"` and `aria-label="월드 콘텐츠 접근성 미러"`.
    3. Inside the wrapper: static `<div>` per post with `id={sr-post-${post.slug}}` — always rendered for AT tree crawling. Each static div contains `<h2>{post.title}</h2>`, `<p>{post.excerpt}</p>`, and one `<p>` per visual alt string.
    4. Below the static divs: a single `<div aria-live="polite" aria-atomic="true">` live region. When `activeSlug` matches a post, announces `"현재 위치: {title}"`. When `activeSlug` is null or 'home', announces `"현재 위치: 월드 홈"` (null-guard for Risk 3 in research).

    **Full component:**

    ```tsx
    'use client'

    import { useWorldStore } from '@/lib/worldStore'

    export interface PostSRData {
      slug: string
      title: string
      excerpt: string
      visuals: { id: string; alt: string }[]
    }

    export default function WorldSRMirror({ posts }: { posts: PostSRData[] }) {
      const activeWaypoint = useWorldStore((s) => s.activeWaypoint)
      const activeSlug = activeWaypoint?.slug ?? null

      const activePost = activeSlug ? posts.find((p) => p.slug === activeSlug) : null
      const liveText = activePost
        ? `현재 위치: ${activePost.title}`
        : '현재 위치: 월드 홈'

      return (
        <div aria-label="월드 콘텐츠 접근성 미러" className="sr-only">
          {/* Static descriptors — always in AT tree for crawling */}
          {posts.map((post) => (
            <div key={post.slug} id={`sr-post-${post.slug}`}>
              <h2>{post.title}</h2>
              <p>{post.excerpt}</p>
              {post.visuals.map((v) => (
                <p key={v.id}>{v.alt}</p>
              ))}
            </div>
          ))}
          {/* aria-live region — announces active waypoint changes only */}
          <div aria-live="polite" aria-atomic="true">
            {liveText}
          </div>
        </div>
      )
    }
    ```

    **Step 2: Make `RootLayout` async** and gather post SR data inside it:

    ```typescript
    export default async function RootLayout({
      children,
    }: {
      children: React.ReactNode
    }) {
      // Gather all post metadata server-side for SR mirror
      const slugs = getPostSlugs()
      const postSRData: PostSRData[] = await Promise.all(
        slugs.map(async (slug) => {
          try {
            const { metadata } = await import(`@/content/posts/${slug}.mdx`)
            const altData = getAltData(slug)
            return {
              slug,
              title: (metadata?.title as string) ?? slug,
              excerpt: (metadata?.excerpt as string) ?? '',
              visuals: altData.visuals,
            }
          } catch {
            return { slug, title: slug, excerpt: '', visuals: [] }
          }
        })
      )
      // ... rest of layout unchanged, add WorldSRMirror inside UIOverlay
    ```

    **Step 3: Add `<WorldSRMirror posts={postSRData} />` inside `<UIOverlay>` as a sibling of `<WorldCursor>`**, positioned before WorldCursor (rendered first in DOM):

    ```tsx
    <UIOverlay>
      <WorldSRMirror posts={postSRData} />
      <WorldCursor />
      <SmoothScrollProviderWrapper>
        <main id="page-content">{children}</main>
      </SmoothScrollProviderWrapper>
    </UIOverlay>
    ```
