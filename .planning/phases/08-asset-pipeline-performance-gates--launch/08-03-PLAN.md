```ts
export function markWorldFirstFrame(): void {
  if (typeof performance === 'undefined') return
  performance.mark('world-first-frame')
  const nav = performance.getEntriesByType('navigation')[0] as PerformanceNavigationTiming
  const delta = performance.now() - (nav?.startTime ?? 0)
  console.log(`[perf] /world first-frame: ${delta.toFixed(1)}ms`)
  try {
    sessionStorage.setItem('world-first-frame', delta.toFixed(1))
  } catch {
    // sessionStorage unavailable (private browsing, etc.)
  }
  try {
    performance.measure('world-first-frame', 'world-first-frame')
  } catch {
    // mark may not exist in all environments
  }
}

// Returns the observer so callers can disconnect() on cleanup
export function observeTextLCP(): PerformanceObserver | null {
  if (typeof PerformanceObserver === 'undefined') return null
  const po = new PerformanceObserver((list) => {
    for (const entry of list.getEntries()) {
      console.log(`[perf] /text LCP: ${entry.startTime.toFixed(1)}ms`)
      try {
        sessionStorage.setItem('text-lcp', entry.startTime.toFixed(1))
      } catch {
        // sessionStorage unavailable
      }
    }
  })
  po.observe({ type: 'largest-contentful-paint', buffered: true })
  return po
}
```

   ```tsx
   // Dev-only draw call monitor for PERF-03 gate
   function DrawCallMonitor() {
     const gl = useThree((s) => s.gl)
     const peakRef = useRef(0)
     useFrame(() => {
       const calls = (gl as any).info?.render?.drawCalls ?? 0
       if (calls > peakRef.current) {
         peakRef.current = calls
         if (typeof sessionStorage !== 'undefined') {
           try {
             sessionStorage.setItem('world-draw-calls-peak', String(calls))
           } catch {
             // sessionStorage unavailable
           }
         }
       }
       if (process.env.NODE_ENV !== 'production' && calls > 600) {
         console.warn(`[perf] WARN draw calls this frame: ${calls}`)
       }
     })
     return null
   }
   ```

Create `app/_perf/page.tsx` as a client component (requires `'use client'` directive). The component reads four keys from `sessionStorage` on mount and renders a pass/fail table for PERF-01 through PERF-04.

The thresholds are:
- PERF-01: `/text/` LCP ≤ 1500 ms (key: `sessionStorage['text-lcp']`)
- PERF-02: `/world` first-frame ≤ 3000 ms (key: `sessionStorage['world-first-frame']`)
- PERF-03: draw call peak ≤ 800 (key: `sessionStorage['world-draw-calls-peak']`)
- PERF-04: mobile poster LCP ≤ 1800 ms — note: PERF-04 cannot be auto-measured from sessionStorage (requires real device + poster image). Display the `text-lcp` value as a proxy or show "measure on real mobile device".

Full file content:
```tsx
'use client'

import { useEffect, useState } from 'react'

interface PerfGate {
  id: string
  label: string
  storageKey: string
  threshold: number
  unit: string
  note?: string
}

const GATES: PerfGate[] = [
  {
    id: 'PERF-01',
    label: '/text/ LCP',
    storageKey: 'text-lcp',
    threshold: 1500,
    unit: 'ms',
  },
  {
    id: 'PERF-02',
    label: '/world first-frame',
    storageKey: 'world-first-frame',
    threshold: 3000,
    unit: 'ms',
  },
  {
    id: 'PERF-03',
    label: 'draw call peak',
    storageKey: 'world-draw-calls-peak',
    threshold: 800,
    unit: 'calls',
  },
  {
    id: 'PERF-04',
    label: 'mobile poster LCP',
    storageKey: 'text-lcp',
    threshold: 1800,
    unit: 'ms',
    note: 'Proxy: /text/ LCP — verify on real mobile device with real poster.jpg',
  },
]

export default function PerfPage() {
  const [values, setValues] = useState<Record<string, string | null>>({})

  useEffect(() => {
    const result: Record<string, string | null> = {}
    for (const gate of GATES) {
      result[gate.storageKey] = sessionStorage.getItem(gate.storageKey)
    }
    setValues(result)
  }, [])

  return (
    <main style={{ fontFamily: 'monospace', padding: '2rem', maxWidth: 640 }}>
      <h1 style={{ fontSize: '1.5rem', marginBottom: '1rem' }}>/_perf — Performance Gates</h1>
      <p style={{ marginBottom: '1.5rem', color: '#666', fontSize: '0.85rem' }}>
        Values are read from sessionStorage. Navigate to /world and /text/ first, then return here.
      </p>
      <table style={{ width: '100%', borderCollapse: 'collapse' }}>
        <thead>
          <tr style={{ borderBottom: '2px solid #ccc' }}>
            <th style={{ textAlign: 'left', padding: '0.5rem' }}>Gate</th>
            <th style={{ textAlign: 'left', padding: '0.5rem' }}>Label</th>
            <th style={{ textAlign: 'right', padding: '0.5rem' }}>Measured</th>
            <th style={{ textAlign: 'right', padding: '0.5rem' }}>Threshold</th>
            <th style={{ textAlign: 'center', padding: '0.5rem' }}>Result</th>
          </tr>
        </thead>
        <tbody>
          {GATES.map((gate) => {
            const raw = values[gate.storageKey]
            const measured = raw !== null && raw !== undefined ? parseFloat(raw) : null
            const passed = measured !== null ? measured <= gate.threshold : null
            return (
              <tr key={gate.id} style={{ borderBottom: '1px solid #eee' }}>
                <td style={{ padding: '0.5rem', fontWeight: 'bold' }}>{gate.id}</td>
                <td style={{ padding: '0.5rem' }}>
                  {gate.label}
                  {gate.note && (
                    <span style={{ display: 'block', fontSize: '0.75rem', color: '#888' }}>
                      {gate.note}
                    </span>
                  )}
                </td>
                <td style={{ padding: '0.5rem', textAlign: 'right' }}>
                  {measured !== null ? `${measured.toFixed(1)} ${gate.unit}` : '—'}
                </td>
                <td style={{ padding: '0.5rem', textAlign: 'right', color: '#666' }}>
                  {`≤ ${gate.threshold} ${gate.unit}`}
                </td>
                <td
                  style={{
                    padding: '0.5rem',
                    textAlign: 'center',
                    fontWeight: 'bold',
                    color: passed === null ? '#999' : passed ? '#16a34a' : '#dc2626',
                  }}
                >
                  {passed === null ? 'N/A' : passed ? 'PASS' : 'FAIL'}
                </td>
              </tr>
            )
          })}
        </tbody>
      </table>
      <p style={{ marginTop: '1.5rem', fontSize: '0.75rem', color: '#888' }}>
        Last updated: {new Date().toISOString()}
      </p>
    </main>
  )
}
```
